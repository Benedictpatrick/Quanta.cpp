# llama.cpp comparison harness.
#   Host (PC check):  cmake -S bench/llamabench -B build-llamabench && cmake --build build-llamabench
#   Android:          scripts/build_llamabench.sh  (three CPU variants -> android/app/src/main/jniLibs)
# llama.cpp is built as its authors recommend for Android (docs/android.md): GGML_NATIVE=OFF, GGML_OPENMP=OFF,
# GGML_LLAMAFILE=OFF; CPU repacking of Q4_0 left ON (default); KleidiAI left at its default (OFF).
cmake_minimum_required(VERSION 3.20)
project(llamabench C CXX)
set(CMAKE_CXX_STANDARD 17)

set(LLAMA_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/llama.cpp)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
if(ANDROID)
  set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
endif()
add_subdirectory(${LLAMA_DIR} llama)

if(ANDROID)
  if(NOT LLAMABENCH_VARIANT)
    message(FATAL_ERROR "set LLAMABENCH_VARIANT (v80 / v82 / v86)")
  endif()
  add_library(llamabench_${LLAMABENCH_VARIANT} SHARED llama_harness.cpp jni.cpp)
  target_link_libraries(llamabench_${LLAMABENCH_VARIANT} PRIVATE llama log)
  target_link_options(llamabench_${LLAMABENCH_VARIANT} PRIVATE -Wl,--gc-sections -Wl,-z,max-page-size=16384)
else()
  add_executable(llamabench llama_harness.cpp host_main.cpp)
  target_link_libraries(llamabench PRIVATE llama)
  if(MINGW)
    target_link_options(llamabench PRIVATE -static)
  endif()
endif()
